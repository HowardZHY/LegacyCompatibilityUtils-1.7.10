package space.libs.asm;

import net.minecraft.launchwrapper.IClassTransformer;
import org.objectweb.asm.*;
import space.libs.asm.visitors.FieldRedirectVisitor;
import space.libs.core.ICoreUtils;

/**
 * @implNote Prepared for possible future usage.
 */
public class FieldRedirectTransformer implements IClassTransformer, ICoreUtils {

    public byte[] transform(String name, String transformedName, byte[] bytes) {
        if (name == null || bytes == null) {
            return bytes;
        }
        if (!name.equals(transformedName) || ClassNameList.ShouldNotTransform(name)) {
            return bytes;
        }
        if (FieldRedirectVisitor.DISABLED) {
            return bytes;
        }
        ClassReader reader = new ClassReader(bytes);
        ClassWriter writer = new ClassWriter(reader, ClassWriter.COMPUTE_MAXS);
        ClassVisitor visitor = new FieldRedirectVisitor(writer);
        reader.accept(visitor, ClassReader.EXPAND_FRAMES);
        return writer.toByteArray();
    }
}
